\subsection{Governing Electromagnetic and Circuit Equations}
\label{subsec:app-eeri2-equations}

\begin{equation}
\label{eq:mean_turn_diameter}
D_{\mathrm{mean}} = \frac{D_{\mathrm{core}} + D_{\mathrm{flange}}}{2}
\end{equation}

\begin{equation}
\label{eq:mean_circumference}
C_{\mathrm{mean}} = \pi D_{\mathrm{mean}}
\end{equation}

\begin{equation}
\label{eq:wire_resistance_cold}
R_{\mathrm{cold}} = \frac{\rho_{\mathrm{Cu}, 20} L}{A_{\mathrm{Cu}}} = \frac{1.72 \times 10^{-8} \times L}{\frac{\pi}{4} d_{\mathrm{Cu}}^2}
\end{equation}

\begin{equation}
\label{eq:temp_resistance}
R(T) = R_{20} \left[ 1 + \alpha_{\mathrm{Cu}} (T - 20^\circ\text{C}) \right], \quad \alpha_{\mathrm{Cu}} = 0.00393\,\mathrm{K}^{-1}
\end{equation}

\begin{equation}
\label{eq:operating_current}
I_{\mathrm{op}} = \frac{V_{\mathrm{boost}}}{R_{\mathrm{series}}(T) + R_{\mathrm{DS(on)}} + R_{\mathrm{harness}}}
\end{equation}

\begin{equation}
\label{eq:magnetomotive_force}
\mathcal{F} = N \cdot I
\end{equation}

\begin{equation}
\label{eq:inductive_energy}
E_L = \frac{1}{2} L I^2
\end{equation}

\begin{equation}
\label{eq:transient_current_rise}
i_L(t) = \frac{V_{\mathrm{boost}}}{R_{\mathrm{loop}}} \left( 1 - e^{-\frac{t}{\tau}} \right), \quad \tau = \frac{L}{R_{\mathrm{loop}}}
\end{equation}

\begin{equation}
\label{eq:differential_opamp_gain}
V_{\mathrm{ISENSE}} = \frac{R_f}{R_{\mathrm{in}}} (V_{\mathrm{ACS}} - V_{\mathrm{ref}}) = 10 \times (V_{\mathrm{ACS}} - 2.50\,\text{V})
\end{equation}

\subsection{Equations}
\label{subsec:c-equations}

\begin{equation}
\label{eq:drum_torque}
\tau_{\mathrm{Drum}} = \mathrm{Load} \times 9.81 \times \frac{\varnothing_{\mathrm{Drum}}}{2 \times 1000} \times K_{\mathrm{Serving\ Factor}}
\end{equation}

\begin{equation}
\label{eq:min_gear}
\mathrm{N_P} = \frac{2\mathrm{k}}{(1 + 2\mathrm{m})\sin^2 \phi} \left( \mathrm{m} + \sqrt{\mathrm{m}^2 + (1 + 2\mathrm{m})\sin^2 \phi} \right)\end{equation}

\begin{equation}
\label{eq:max_gear}
\mathrm{N_G} = \frac{\mathrm{N_P}^2 \sin^2 \phi - 4k^2}{4k - 2N_P \sin^2 \phi}
\end{equation}

\begin{equation}
\label{eq:pitch_circle}
D_P = \mathrm{module} \times \mathrm{z}
\end{equation}

\begin{equation}
\label{eq:gear_module}
\mathrm{D} = \frac{p}{\pi}
\end{equation}


\begin{equation}
\label{eq:addendum_circle}
D = D_p + 2 \times m
\end{equation}


\begin{equation}
\label{eq:root_circle}
D_r = D_p - 2 \times b
\end{equation}


\begin{equation}
\label{eq:dedendum}
b = 1.25 \times m
\end{equation}

\begin{equation}
\label{eq:clearance_circle}
D_c = D_p - 2\times m
\end{equation}


\begin{equation}
\label{eq:circular_pitch}
p = m\times\pi
\end{equation}

\begin{equation}
\label{eq:base_circle}
D_b = D_p \cos{\alpha}
\end{equation}

\begin{equation}
    \label{eq:rig_torque}
    \tau = F \times d
\end{equation}

\textbf{Subsystem function:} Raise and lower a payload of up to \InMassRated\,kg through
a \InStroke\,m stroke on a rope and drum, hold it without drift when power or command is
lost, and report the on-hook load to the controller (FR4, FR5, FR6, PT2).
 
\textbf{Test setup:} Complete hoist assembly (motor, coupling, gearbox, brake, drum, rope,
load cell with sheave, HX711) bolted to the test frame; VSD or bench supply for the motor;
48\,V brake supply and relay; controller or bench read-out of the HX711; calibrated
reference masses (5, 10, 20, 30\,kg); stopwatch or encoder for timing; tape measure for drift.
Protect the area below the load. The 30\,kg proof test is performed last.
