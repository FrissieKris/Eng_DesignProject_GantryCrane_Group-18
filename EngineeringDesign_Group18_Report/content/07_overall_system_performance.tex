\begin{lstlisting}[language=Python,
                   caption={Sensor initialisation routine.},
                   label=lst:sensor-init]
import sensor_library as sl

def init_sensor(pin: int, baud: int = 9600) -> sl.Sensor:
    """Initialise and calibrate the sensor on the given GPIO pin."""
    sensor = sl.Sensor(pin=pin, baud=baud)
    sensor.calibrate()
    return sensor
\end{lstlisting}
